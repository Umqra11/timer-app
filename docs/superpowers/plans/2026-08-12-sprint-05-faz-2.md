- [ ] **Step 3: Write `functions/src/presence.ts` — server-side presence write**

```typescript
import { onCall, HttpsError } from 'firebase-functions/v2/https';
import { getFirestore, FieldValue, Timestamp } from 'firebase-admin/firestore';

const db = getFirestore();

const RATE_PER_MINUTE = 5;
const RATE_PER_HOUR = 30;

export const onPresenceChange = onCall(async (request) => {
  const { roomId, status, elapsedMs } = request.data as {
    roomId: string; status: 'running' | 'paused' | 'finished' | 'idle'; elapsedMs: number;
  };
  const uid = request.auth?.uid ?? request.data.uid; // MVP: device-uid via data
  if (!uid) throw new HttpsError('unauthenticated', 'uid required');

  // Rate limit
  const rateRef = db.doc(`users/${uid}/rateLimit/presence`);
  const rate = await rateRef.get();
  const now = Date.now();
  const data = rate.data() ?? { minuteCount: 0, minuteResetAt: now + 60_000, hourCount: 0, hourResetAt: now + 3_600_000 };
  const minuteReset = now >= data.minuteResetAt ? now + 60_000 : data.minuteResetAt;
  const hourReset = now >= data.hourResetAt ? now + 3_600_000 : data.hourResetAt;
  if (data.minuteCount >= RATE_PER_MINUTE || data.hourCount >= RATE_PER_HOUR) {
    throw new HttpsError('resource-exhausted', 'rate limit');
  }
  await rateRef.set({
    minuteCount: (data.minuteCount ?? 0) + 1,
    minuteResetAt: minuteReset,
    hourCount: (data.hourCount ?? 0) + 1,
    hourResetAt: hourReset
  });

  await db.doc(`rooms/${roomId}/presence/${uid}`).set({
    uid, status, elapsedMs, updatedAt: FieldValue.serverTimestamp()
  }, { merge: true });

  return { ok: true };
});
```

```typescript
export const onReactionSend = onCall(async (request) => {
  const { roomId, targetUid, text, senderUsername } = request.data as { roomId: string; targetUid: string; text: string; senderUsername: string };
  const uid = request.auth?.uid ?? request.data.uid;
  if (!uid) throw new HttpsError('unauthenticated', 'uid required');
  if (!text || text.length > 60) throw new HttpsError('invalid-argument', 'text required');
  if (uid === targetUid) throw new HttpsError('failed-precondition', 'self-target');

  // Rate limit (reuse logic from presence.ts — refactor to shared util)
  await assertRateLimit(uid, 'reactions');

  // Self-target already checked above; rate limit passed
  const id = crypto.randomUUID();
  await db.doc(`rooms/${roomId}/reactions/${id}`).set({
    targetUid, senderUid: uid, senderUsername,
    text, createdAt: FieldValue.serverTimestamp(),
    expireAt: Timestamp.fromMillis(Date.now() + 4 * 60 * 60 * 1000) // 4h TTL
  });
  return { ok: true, id };
});
```

- [ ] **Step 2: Write `functions/src/rooms.ts` — owner-checked delete**

```typescript
export const onRoomDelete = onCall(async (request) => {
  const { roomId } = request.data as { roomId: string };
  const uid = request.auth?.uid ?? request.data.uid;
  if (!uid) throw new HttpsError('unauthenticated', 'uid required');
  const room = await db.doc(`rooms/${roomId}`).get();
  if (!room.exists) throw new HttpsError('not-found', 'room not found');
  if (room.data()?.ownerUid !== uid) throw new HttpsError('permission-denied', 'only owner can delete');
  await db.doc(`rooms/${roomId}`).delete(); // Part 3 will add recursive cleanup
  return { ok: true };
});
```

```typescript
async function recursiveDelete(roomId: string) {
  const subs = ['presence', 'reactions'];
  for (const sub of subs) {
    const snap = await db.collection(`rooms/${roomId}/${sub}`).get();
    const batch = db.batch();
    snap.docs.forEach(d => batch.delete(d.ref));
    await batch.commit();
  }
  // Delete joinedRooms for all members
  const presence = await db.collection(`rooms/${roomId}/presence`).get();
  const memberUids = presence.docs.map(d => d.id);
  for (const uid of memberUids) {
    await db.doc(`users/${uid}/joinedRooms/${roomId}`).delete();
  }
  // Finally delete the room itself
  await db.doc(`rooms/${roomId}`).delete();
}
```
