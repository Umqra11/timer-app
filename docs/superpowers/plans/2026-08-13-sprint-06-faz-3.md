- [ ] **Step 2: Add the heartbeat $effect block**

```ts
$effect(() => {
  if (status !== 'running') return;

  const intervalId = setInterval(() => {
    pushToRemote();
  }, 60_000);

  const onVisibilityChange = () => {
    if (document.visibilityState === 'visible') {
      pushToRemote();
    }
  };
  document.addEventListener('visibilitychange', onVisibilityChange);

  return () => {
    clearInterval(intervalId);
    document.removeEventListener('visibilitychange', onVisibilityChange);
  };
});
```

**Note:** `status` must be referenced inside the effect for Svelte 5 reactivity tracking. If `status` is currently a closure variable, ensure it's `$state(...)` so the effect re-runs on status change.

## Task 7 — P1: Implement cleanup function (GREEN)

**Files:**
- Create: `mobile/functions/src/cleanupPresence.ts`

**Consumes:** Task 6 tests (currently failing).

**Produces:**
- `cleanupIdlePresence` scheduled function: every 6 hours, batch-update `presence/{uid}` docs where `lastHeartbeatAt < now - 24h` AND `status === 'running'` → `status: 'idle'`.

- [ ] **Step 1: Write the implementation**

Create `mobile/functions/src/cleanupPresence.ts`:

```ts
import { onSchedule } from 'firebase-functions/v2/scheduler';
import { getFirestore } from 'firebase-admin/firestore';
import { logger } from 'firebase-functions';

const TWENTY_FOUR_HOURS_MS = 24 * 60 * 60 * 1000;

export const cleanupIdlePresence = onSchedule(
  {
    schedule: 'every 6 hours',
    timeZone: 'UTC',
    region: 'us-central1',
  },
  async () => {
    const db = getFirestore();
    const cutoff = Date.now() - TWENTY_FOUR_HOURS_MS;

    const snapshot = await db
      .collection('presence')
      .where('lastHeartbeatAt', '<', cutoff)
      .where('status', '==', 'running')
      .get();

    if (snapshot.empty) {
      logger.info('cleanupIdlePresence: no stale docs', { cutoff });
      return;
    }

    const batch = db.batch();
    snapshot.docs.forEach((doc) => {
      batch.update(doc.ref, { status: 'idle' });
    });
    await batch.commit();

    logger.info('cleanupIdlePresence: marked idle', {
      updated: snapshot.size,
      cutoff,
    });
  }
);
```
